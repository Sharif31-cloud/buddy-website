def fetch_ohlc_mt5(symbol, timeframe, bars=FETCH_COUNT):
    """
    Fetch OHLC data from MT5 for a single symbol.
    Returns a pandas DataFrame indexed by datetime, or None if no data.
    """
    if mt5 is None:
        logger.error("MT5 library not available.")
        return None

    # Ensure symbol is selected in Market Watch
    if not mt5.symbol_select(symbol, True):
        logger.warning(f"Could not select {symbol} in Market Watch.")
        return None

    rates = mt5.copy_rates_from_pos(symbol, timeframe, 0, bars)
    if rates is None or len(rates) == 0:
        logger.warning(f"MT5 returned no data for {symbol}")
        return None

    df = pd.DataFrame(rates)
    df['time'] = pd.to_datetime(df['time'], unit='s')
    df.set_index('time', inplace=True)
    df = df.rename(columns={
        'open': 'open',
        'high': 'high',
        'low': 'low',
        'close': 'close',
        'tick_volume': 'volume'
    })
    return df[['open', 'high', 'low', 'close', 'volume']]

def add_indicators(df, adx_period=14):
    df = df.copy()

    # --- Moving Averages ---
    df["SMA_5"] = df["close"].rolling(5).mean()
    df["EMA_10"] = df["close"].ewm(span=10, adjust=False).mean()
    df["EMA_20"] = df["close"].ewm(span=20, adjust=False).mean()

    # --- RSI ---
    delta = df["close"].diff()
    gain = delta.where(delta > 0, 0).rolling(7).mean()
    loss = -delta.where(delta < 0, 0).rolling(7).mean()
    rs = gain / loss
    df["RSI_7"] = 100 - (100 / (1 + rs))

    # --- MACD ---
    df["EMA_8"] = df["close"].ewm(span=8, adjust=False).mean()
    df["EMA_21"] = df["close"].ewm(span=21, adjust=False).mean()
    df["MACD"] = df["EMA_8"] - df["EMA_21"]
    df["MACD_Signal"] = df["MACD"].ewm(span=5, adjust=False).mean()

    # --- Stochastic ---
    low7 = df["low"].rolling(7).min()
    high7 = df["high"].rolling(7).max()
    df["Stochastic_K"] = ((df["close"] - low7) / (high7 - low7)) * 100
    df["Stochastic_D"] = df["Stochastic_K"].rolling(3).mean()

    # --- ATR ---
    df["H-L"] = df["high"] - df["low"]
    df["H-PC"] = (df["high"] - df["close"].shift(1)).abs()
    df["L-PC"] = (df["low"] - df["close"].shift(1)).abs()
    df["TR"] = df[["H-L", "H-PC", "L-PC"]].max(axis=1)
    df["ATR_7"] = df["TR"].rolling(7).mean()
    df["ATR_MA_14"] = df["ATR_7"].rolling(14).mean()
    df.drop(columns=["H-L", "H-PC", "L-PC"], inplace=True)

    # --- ADX ---
    high = df["high"]
    low = df["low"]
    close = df["close"]

    # True Range
    tr = pd.Series(np.maximum.reduce([
        high - low,
        (high - close.shift(1)).abs(),
        (low - close.shift(1)).abs()
    ]), index=df.index)
    atr_adx = tr.rolling(adx_period, min_periods=1).mean()

    # Directional Movement
    up = high.diff()
    down = -low.diff()
    plus_dm = np.where((up > down) & (up > 0), up, 0.0)
    minus_dm = np.where((down > up) & (down > 0), down, 0.0)

    plus_di = 100 * pd.Series(plus_dm, index=df.index).rolling(adx_period, min_periods=1).sum() / atr_adx.replace(0,
                                                                                                                  1e-10)
    minus_di = 100 * pd.Series(minus_dm, index=df.index).rolling(adx_period, min_periods=1).sum() / atr_adx.replace(0,
                                                                                                                    1e-10)

    dx = 100 * (plus_di - minus_di).abs() / (plus_di + minus_di).replace(0, 1e-10)
    df["ADX_14"] = dx.rolling(adx_period, min_periods=1).mean()

    return df

def technical_signal(row, prev_row=None):
    """
    Combined logic:
    - Fast: reacts in 1–2 candles
    - Safe: avoids chop & false reversals
    - Early: enters before indicators fully lag
    """

    score = 0

    # =========================
    # 1. TREND STRUCTURE (CORE)
    # =========================
    # This PREVENTS selling in uptrends
    if pd.notna(row.get("EMA_10")) and pd.notna(row.get("EMA_20")):
        if row["EMA_10"] > row["EMA_20"]:
            score += 30
        elif row["EMA_10"] < row["EMA_20"]:
            score -= 30

    # Trend slope confirmation (early, not laggy)
    if prev_row is not None and pd.notna(row.get("EMA_10")):
        if row["EMA_10"] > prev_row["EMA_10"]:
            score += 10
        elif row["EMA_10"] < prev_row["EMA_10"]:
            score -= 10

    # =========================
    # 2. MOMENTUM (EARLY ENTRY)
    # =========================
    # RSI zone + direction
    if pd.notna(row.get("RSI_7")):
        if row["RSI_7"] > 55:
            score += 15
        elif row["RSI_7"] < 45:
            score -= 15

        if prev_row is not None:
            if row["RSI_7"] > prev_row["RSI_7"]:
                score += 5
            elif row["RSI_7"] < prev_row["RSI_7"]:
                score -= 5

    # MACD momentum (not cross-only → faster)
    if pd.notna(row.get("MACD")) and pd.notna(row.get("MACD_Signal")):
        if row["MACD"] > row["MACD_Signal"]:
            score += 15
        elif row["MACD"] < row["MACD_Signal"]:
            score -= 15

        if prev_row is not None:
            if row["MACD"] > prev_row["MACD"]:
                score += 5
            elif row["MACD"] < prev_row["MACD"]:
                score -= 5

    # =========================
    # 3. PRICE ACTION (FAST)
    # =========================
    body = abs(row["close"] - row["open"])
    range_ = row["high"] - row["low"]

    if range_ > 0:
        body_ratio = body / range_

        # Strong candle
        if body_ratio > 0.6:
            if row["close"] > row["open"]:
                score += 10
            elif row["close"] < row["open"]:
                score -= 10

    # =========================
    # 4. CHOPPY MARKET FILTER
    # =========================
    # Reduce confidence instead of flipping direction
    if pd.notna(row.get("ATR_7")) and pd.notna(row.get("ATR_MA_14")):
        if row["ATR_7"] < row["ATR_MA_14"] * 0.8:
            score *= 0.6  # chop → weaken signals

    # =========================
    # 5. FINAL DECISION
    # =========================
    if score >= 35:
        return "BUY"
    elif score <= -35:
        return "SELL"
    else:
        return "HOLD"

def trend_filter_1h(df_60m):
    if df_60m is None or len(df_60m) < 200:
        return None  # not enough data

    df_60m["EMA200"] = df_60m["close"].ewm(span=200).mean()

    last = df_60m.iloc[-1]

    if last["close"] > last["EMA200"]:
        return "BUY"
    elif last["close"] < last["EMA200"]:
        return "SELL"
    else:
        return None

def detect_liquidity_sweep(df):
    if df is None or len(df) < 3:
        return None

    prev = df.iloc[-2]
    last = df.iloc[-1]

    # 🔴 SELL sweep (grab buy-side liquidity)
    if last["high"] > prev["high"] and last["close"] < prev["high"]:
        return "SELL"

    # 🟢 BUY sweep (grab sell-side liquidity)
    if last["low"] < prev["low"] and last["close"] > prev["low"]:
        return "BUY"

    return None

# FINAL SIGNAL LOGIC (NEW, STABLE)
# -----------------------------
def final_signal_logic(
    tech_5m,
    tech_15m,
    last_price,
    support_low,
    support_high,
    resistance_low,
    resistance_high,
    df_15m,
    df_60m=None,
    liquidity_signal=None,     # ✅ ADD
    log_callback=None          # ✅ ADD (optional)
):
    """
    PRO FINAL SIGNAL LOGIC (UPGRADED)

    + 1H trend filter (EMA 200)
    """

    # =========================
    # SAFETY CHECK
    # =========================
    if df_15m is None or len(df_15m) < 10:
        return "HOLD", 0

    if tech_5m == "HOLD" or tech_15m == "HOLD":
        return "HOLD", 0

    if tech_5m != tech_15m:
        return "HOLD", 0

    # =========================
    # 🔥 1H TREND FILTER
    # =========================
    if df_60m is not None and len(df_60m) >= 200:
        df_60m["EMA200"] = df_60m["close"].ewm(span=200).mean()
        last_1h = df_60m.iloc[-1]

        trend_1h = None
        if last_1h["close"] > last_1h["EMA200"]:
            trend_1h = "BUY"
        elif last_1h["close"] < last_1h["EMA200"]:
            trend_1h = "SELL"

        if trend_1h and tech_5m != trend_1h:
            return "HOLD", 0
    else:
        return "HOLD", 0

    # =========================
    # MARKET DATA
    # =========================
    last = df_15m.iloc[-1]
    prev = df_15m.iloc[-2]

    prev_close = prev["close"]

    atr = df_15m["ATR_7"].iloc[-1]
    zone_buffer = atr * 0.2

    # =========================
    # BREAKOUT
    # =========================
    is_breakout_buy = (
        tech_5m == "BUY"
        and prev_close >= resistance_low
        and last_price >= resistance_high * 0.999
    )

    is_breakout_sell = (
        tech_5m == "SELL"
        and prev_close <= support_high
        and last_price <= support_low * 1.001
    )

    # =========================
    # ZONE PROTECTION
    # =========================
    if tech_5m == "BUY":
        if not is_breakout_buy and last_price >= (resistance_high - zone_buffer):
            return "HOLD", 0

    if tech_5m == "SELL":
        if not is_breakout_sell and last_price <= (support_low + zone_buffer):
            return "HOLD", 0

    # =========================
    # 🔥 CANDLE CONFIRMATION
    # =========================
    body = abs(last["close"] - last["open"])
    range_candle = last["high"] - last["low"]

    if range_candle == 0:
        return "HOLD", 0

    body_ratio = body / range_candle

    bullish_strong = (
        last["close"] > last["open"] and
        body_ratio > 0.5 and
        last["close"] > prev["high"]
    )

    bearish_strong = (
        last["close"] < last["open"] and
        body_ratio > 0.5 and
        last["close"] < prev["low"]
    )

    # =========================
    # APPLY CONFIRMATION
    # =========================
    if tech_5m == "BUY":
        if not (bullish_strong or is_breakout_buy):
            return "HOLD", 0

    elif tech_5m == "SELL":
        if not (bearish_strong or is_breakout_sell):
            return "HOLD", 0

    # =========================
    # ✅ FIX: DEFINE STRONG_CANDLE
    # =========================
    strong_candle = (
        bullish_strong if tech_5m == "BUY"
        else bearish_strong if tech_5m == "SELL"
        else False
    )

    # =========================
    # PULLBACK / REJECTION
    # =========================
    if tech_5m == "BUY" and not is_breakout_buy:
        pullback = prev["close"] > last["close"]
        if not (pullback or strong_candle):
            return "HOLD", 0

    elif tech_5m == "SELL" and not is_breakout_sell:
        pullback = prev["close"] < last["close"]
        if not (pullback or strong_candle):
            return "HOLD", 0

    # =============================
    # 🔥 LIQUIDITY SWEEP FILTER
    # =============================
    if liquidity_signal is not None:
        if tech_5m != liquidity_signal:
            if log_callback:
                log_callback(
                    f"🚫 Blocked {tech_5m} | Liquidity sweep says {liquidity_signal}"
                )
            return "HOLD", 0

    # =========================
    # POSITION FILTER
    # =========================
    range_total = resistance_high - support_low

    if range_total > 0:
        position = (last_price - support_low) / range_total
        position = max(0, min(1, position))

        if tech_5m == "BUY" and not is_breakout_buy and position > 0.7:
            return "HOLD", 0

        if tech_5m == "SELL" and not is_breakout_sell and position < 0.3:
            return "HOLD", 0

    # =========================
    # STRENGTH
    # =========================
    strength = 70

    if tech_5m == "BUY":
        if is_breakout_buy:
            strength = 85
        else:
            distance = resistance_high - last_price
            if distance > atr:
                strength += 10
            if distance < atr * 0.5:
                strength -= 20

    elif tech_5m == "SELL":
        if is_breakout_sell:
            strength = 85
        else:
            distance = last_price - support_low
            if distance > atr:
                strength += 10
            if distance < atr * 0.5:
                strength -= 20

    strength = max(0, min(100, strength))

    if strength < 30:
        return "HOLD", strength

    return tech_5m, strength

def pip_value_and_pip(MT5_symbol):
    """
    Returns the pip value and contract size for a given MT5 symbol.
    Works for BTC, XAU, XAG, and standard Forex pairs.
    """
    pair = MT5_symbol.upper()

    if "BTC" in pair:
        pip = 0.01  # 1 pip = $0.01
        contract_size = 1  # 1 lot = 1 BTC
    elif "XAU" in pair:
        pip = 0.01  # 1 pip = $0.01
        contract_size = 1  # 1 lot = 1 oz
    elif "XAG" in pair:
        pip = 0.001  # 1 pip = $0.001
        contract_size = 100  # 1 lot = 100 oz (adjust if your broker uses different contract size)
    else:
        pip = 0.0001  # standard Forex pip
        contract_size = 100000.0

    return pip, contract_size

def risk_filters_ok(tick, atr_15m, MT5_symbol):
    if tick is None or atr_15m is None:
        return False

    # -----------------------------
    # ATR minimums (market sanity)
    # -----------------------------
    if "XAU" in MT5_symbol.upper():
        atr_min = 3.5
        spread_mult = 0.35
        min_spread_cap = 0.15
    elif "XAG" in MT5_symbol.upper():
        atr_min = 0.2
        spread_mult = 0.35
        min_spread_cap = 0.03
    elif "BTC" in MT5_symbol.upper() or "ETH" in MT5_symbol.upper():
        atr_min = 40.0
        spread_mult = 0.30
        min_spread_cap = atr_15m * 0.20
    else:
        # Forex
        atr_min = 0.0004
        spread_mult = 0.7
        min_spread_cap = 0.00025  # hard floor

    # -----------------------------
    # Spread filter (adaptive + floor)
    # -----------------------------
    spread = tick.ask - tick.bid
    max_allowed_spread = max(atr_15m * spread_mult, min_spread_cap)

    if spread > max_allowed_spread:
        logger.warning(
            f"Spread too high for {MT5_symbol}: "
            f"{spread:.5f} > max({atr_15m:.5f} * {spread_mult}, {min_spread_cap})"
        )
        return False

    # -----------------------------
    # ATR filter
    # -----------------------------
    if atr_15m < atr_min:
        logger.warning(
            f"ATR too low for {MT5_symbol}: "
            f"ATR={atr_15m:.5f}, min={atr_min:.5f}"
        )
        return False

    return True

# Calculate lot size from risk %
def calculate_lot_size_from_risk(sl_price, entry_price, balance, symbol, risk_percent=1.0):
    """
    Risk-based lot size calculation.
    sl_price: stoploss price
    entry_price: entry price
    balance: account balance
    symbol: MT5 symbol
    risk_percent: % of balance to risk (default = 1%)

    Special rule: USTEC trades use at least 0.05 lots.
    """

    import MetaTrader5 as mt5

    # Amount to risk
    risk_amount = balance * (risk_percent / 100)

    # Distance from entry to SL in price units
    sl_distance = abs(entry_price - sl_price)
    if sl_distance <= 0:
        return 0.01  # fallback minimum lot

    # Get symbol info
    info = mt5.symbol_info(symbol)
    if info is None:
        return 0.01

    tick_size = info.trade_tick_size
    tick_value = info.trade_tick_value
    if tick_size == 0 or tick_value == 0:
        return 0.01

    # Value per 1 lot per 1 price move
    value_per_point_per_lot = tick_value / tick_size

    # Potential loss per 1 lot
    loss_per_lot = sl_distance * value_per_point_per_lot
    if loss_per_lot <= 0:
        return 0.01

    # Calculate lot size
    lots = risk_amount / loss_per_lot

    # Apply broker min/max
    lots = max(info.volume_min, min(lots, info.volume_max))

    # Normalize to broker step
    step = info.volume_step
    lots = round(lots / step) * step

    MIN_LOT_OVERRIDES = {
        # Crypto + Indices
        "USTEC": 0.05,
        "BTCUSD": 0.05,
        "ETHUSD": 0.05,


        # Forex
        "EURUSD": 0.02,
        "EURJPY": 0.02,
        "GBPUSD": 0.02,
        "GBPJPY": 0.02,
        "USOIL": 0.02,
    }

    symbol_upper = symbol.upper()

    for key, min_lot in MIN_LOT_OVERRIDES.items():
        if key in symbol_upper and lots < min_lot:
            lots = min_lot
            break

    return lots

def calculate_sl_tp(row_5m, df_15m, signal, symbol):
    import MetaTrader5 as mt5

    atr = df_15m["ATR_7"].iloc[-1]
    entry = float(row_5m["close"])

    info = mt5.symbol_info(symbol)
    if info is None:
        return None, None

    digits = info.digits
    point = info.point
    stop_level = info.trade_stops_level * point  # broker minimum distance

    # 🚫 Safety: invalid ATR
    if atr is None or atr <= 0:
        return None, None

    # ✅ Ensure minimum SL size (avoid tiny forex ATR)
    min_sl = 10 * point   # ~10 pips safety floor
    SL_points = max(atr * 2.0, min_sl)

    RR = 2.5

    # -------------------------
    # CALCULATE RAW SL/TP
    # -------------------------
    if signal == "BUY":
        sl = entry - SL_points
        tp = entry + SL_points * RR

    elif signal == "SELL":
        sl = entry + SL_points
        tp = entry - SL_points * RR

    else:
        return None, None

    # -------------------------
    # ENFORCE BROKER STOP LEVEL
    # -------------------------
    if abs(entry - sl) < stop_level:
        if signal == "BUY":
            sl = entry - stop_level
        else:
            sl = entry + stop_level

    if abs(entry - tp) < stop_level:
        if signal == "BUY":
            tp = entry + stop_level
        else:
            tp = entry - stop_level

    # -------------------------
    # FINAL ROUNDING (CORRECT)
    # -------------------------
    sl = round(sl, digits)
    tp = round(tp, digits)

    return sl, tp

def mt5_init_and_login(login: int, password: str, server: str):
    if mt5 is None:
        raise RuntimeError("MetaTrader5 package not installed.")
    if not mt5.initialize():
        print("mt5.initialize() failed, error:", mt5.last_error())
        return False
    ok = mt5.login(login, password, server)
    if not ok:
        print("MT5 login failed:", mt5.last_error())
        return False
    print("MT5 logged in:", login, "server:", server)
    return True


def ensure_symbol_ready(symbol):
    if mt5 is None:
        return False, "mt5 not installed"
    info = mt5.symbol_info(symbol)
    if info is None:
        return False, f"{symbol} not found in MT5"
    if not info.visible:
        mt5.symbol_select(symbol, True)
    return True, None

def is_special_asset(symbol):
    s = symbol.upper()
    return any(x in s for x in ["XAU", "XAG", "BTC", "USOIL"])


def compute_momentum(df):
    if df is None or len(df) < 10:
        return 0.0

    close = df["close"]
    momentum_raw = (close.iloc[-1] - close.iloc[-5]) / close.iloc[-5]
    return abs(momentum_raw)


def get_momentum_level(symbol, df):
    m = compute_momentum(df)

    s = symbol.upper()

    if "BTC" in s:
        return "high" if m > 0.003 else "low"
    elif "XAU" in s or "XAG" in s:
        return "high" if m > 0.002 else "low"
    else:
        return "high" if m > 0.0015 else "low"

                    else:
                        BE_TRIGGER = 1
                        TRAIL_TRIGGER = 2.0

                        # ✅ Forex-safe buffers (in price, not USD)
                        BE_BUFFER = 20 * point  # ~2 pips
                        TRAIL_STEP = 100 * point  # ~10 pips

                    # ======================================================
                    # 🟢 STEP 1: BREAKEVEN (SAFE + SPREAD AWARE)
                    # ======================================================
                    if profit_usd >= BE_TRIGGER:

                        spread = abs(tick.ask - tick.bid)

                        if is_buy:
                            be_sl = entry + BE_BUFFER + spread
                            log(f"BE DEBUG | {symbol} profit={profit_usd:.2f} entry={entry} be_sl={be_sl} base_sl={base_sl}")

                            if base_sl is None or be_sl > base_sl:
                                base_sl = be_sl

                            be_sl = entry - BE_BUFFER - spread
                            log(f"BE DEBUG | {symbol} profit={profit_usd:.2f} entry={entry} be_sl={be_sl} base_sl={base_sl}")

                            if base_sl is None or be_sl < base_sl:
                                base_sl = be_sl

                    # ======================================================
                    # 🔵 STEP 2: TRAILING (CLEAN & EFFECTIVE)
                    # ======================================================
                    if profit_usd >= TRAIL_TRIGGER:

                        if is_buy:
                            candidate = price - TRAIL_STEP

                            log(f"TRAIL DEBUG | {symbol} price={price} candidate={candidate} base_sl={base_sl}")

                            if base_sl is None or candidate > base_sl:
                                base_sl = candidate

                        else:
                            candidate = price + TRAIL_STEP

                            log(f"TRAIL DEBUG | {symbol} price={price} candidate={candidate} base_sl={base_sl}")

                            if base_sl is None or candidate < base_sl:
                                base_sl = candidate

                    # ======================================================
                    # 🧱 FINAL SL DECISION
                    # ======================================================
                    new_sl = base_sl

                    if new_sl is None:
                        continue

                    # ======================================================
                    # 🚫 VALIDATION (STRICT)
                    # ======================================================
                    if is_buy and current_sl != 0 and new_sl <= current_sl:
                        continue

                    if not is_buy and current_sl != 0 and new_sl >= current_sl:
                        continue

                    # ======================================================
                    # 🛡️ BROKER SAFETY CHECK
                    # ======================================================
                    if is_buy:
                        if new_sl > price - stop_level:
                            continue
                    else:
                        if new_sl < price + stop_level:
                            continue

                    # ======================================================
                    # 🔁 AVOID DUPLICATE UPDATES
                    # ======================================================
                    if current_sl != 0 and round(new_sl, digits) == round(current_sl, digits):
                        continue

                    # ======================================================
                    # 🚀 APPLY SL
                    # ======================================================
                    request = {
                        "action": mt5.TRADE_ACTION_SLTP,
                        "position": pos.ticket,
                        "sl": round(new_sl, digits),
                        "tp": pos.tp
                    }

                    res = mt5.order_send(request)

                    if res is None:

                        continue

                    if res.retcode != mt5.TRADE_RETCODE_DONE:

                        f"{current_sl:.5f} → {new_sl:.5f} | profit={profit_usd:.2f}")

                except Exception as e:
                    log(f"❌ error ticket={pos.ticket}: {e}")

        except Exception as e:
            log(f"❌ monitor error {symbol}: {e}")

        time.sleep(interval)


def calculate_sr_levels(df_15m, lookback=50):
    """
    Returns support and resistance ZONES (ranges)
    """

    if df_15m is None or len(df_15m) < lookback:
        return None, None, None, None

    highs = df_15m["high"].iloc[-lookback:]
    lows = df_15m["low"].iloc[-lookback:]

    # Resistance zone (top area where price rejects)
    resistance_low = highs.quantile(0.85)
    resistance_high = highs.quantile(0.95)

    # Support zone (bottom area where price bounces)
    support_low = lows.quantile(0.05)
    support_high = lows.quantile(0.15)

    return support_low, support_high, resistance_low, resistance_high


def calculate_choppy_thresholds(df_5m, df_15m, atr_mult_pct=0.75, adx_mult_pct=0.6, atr_period=20, adx_period=14):
    """
    Calculate dynamic ATR and ADX thresholds for choppy market detection.
    Returns:
        atr_thresh_5m, atr_thresh_15m, adx_thresh_5m, adx_thresh_15m
    """
    df_5m = add_indicators(df_5m)
    df_15m = add_indicators(df_15m)

    # ATR thresholds
    atr_avg_5m = df_5m["ATR_7"].rolling(atr_period).mean().iloc[-1]
    atr_avg_15m = df_15m["ATR_7"].rolling(atr_period).mean().iloc[-1]
    atr_thresh_5m = atr_mult_pct * atr_avg_5m
    atr_thresh_15m = atr_mult_pct * atr_avg_15m

    # ADX thresholds
    def calc_adx(df, period=adx_period):
        df = df.copy()
        df['H-L'] = df['high'] - df['low']
        df['H-PC'] = (df['high'] - df['close'].shift(1)).abs()
        df['L-PC'] = (df['low'] - df['close'].shift(1)).abs()
        df['TR'] = df[['H-L', 'H-PC', 'L-PC']].max(axis=1)
        df['ATR'] = df['TR'].rolling(period).mean()

        df['+DM'] = df['high'].diff()
        df['-DM'] = -df['low'].diff()
        df['+DM'] = df['+DM'].where((df['+DM'] > df['-DM']) & (df['+DM'] > 0), 0.0)
        df['-DM'] = df['-DM'].where((df['-DM'] > df['+DM']) & (df['-DM'] > 0), 0.0)

        df['+DI'] = 100 * df['+DM'].rolling(period).sum() / df['ATR']
        df['-DI'] = 100 * df['-DM'].rolling(period).sum() / df['ATR']
        df['DX'] = 100 * (df['+DI'] - df['-DI']).abs() / (df['+DI'] + df['-DI'])
        adx = df['DX'].rolling(period).mean()
        return adx.iloc[-1]

    adx_thresh_5m = adx_mult_pct * calc_adx(df_5m)
    adx_thresh_15m = adx_mult_pct * calc_adx(df_15m)

    return atr_thresh_5m, atr_thresh_15m, adx_thresh_5m, adx_thresh_15m

def drawdown_ok() -> bool:
    global PEAK_EQUITY, DAILY_START_EQUITY, LAST_EQUITY_DAY

    acc = mt5.account_info()
    if acc is None:
        return True

    equity = float(acc.equity)
    today = datetime.utcnow().date()

    # -----------------------------
    # 🔹 DAILY RESET
    # -----------------------------
    if LAST_EQUITY_DAY != today:
        DAILY_START_EQUITY = equity
        LAST_EQUITY_DAY = today
        logger.info(f"🌅 New trading day → Daily equity reset: {equity:.2f}")

    # -----------------------------
    # 🔹 UPDATE PEAK EQUITY
    # -----------------------------
    if equity > PEAK_EQUITY:
        PEAK_EQUITY = equity

    # -----------------------------
    # 🔹 CALCULATIONS
    # -----------------------------
    total_dd = ((PEAK_EQUITY - equity) / PEAK_EQUITY) * 100
    daily_dd = ((DAILY_START_EQUITY - equity) / DAILY_START_EQUITY) * 100

    # -----------------------------
    # 🔴 HARD STOP CONDITIONS
    # -----------------------------
    if total_dd >= MAX_TOTAL_DRAWDOWN_PCT:
        logger.critical(f"⛔ TOTAL DD HIT: {total_dd:.2f}%")
        close_all_positions()
        cooldown_until_all = time.time() + 3600  # pause 1 hour

    if daily_dd >= MAX_DAILY_DRAWDOWN_PCT:
        logger.critical(f"⛔ DAILY DD HIT: {daily_dd:.2f}%")



    # -----------------------------
    # ✅ SAFE
    # -----------------------------
    logger.info(f"DD OK | Total: {total_dd:.2f}% | Daily: {daily_dd:.2f}%")
    return True

def performance_metrics(trades: pd.DataFrame) -> dict:
    """
    Robust performance metrics.
    Handles extra columns (order_action, order_result) safely.
    """

    if trades.empty:
        return {}

    # -------- REQUIRED COLUMNS CHECK --------
    required_cols = {"time", "account_balance", "account_equity"}
    if not required_cols.issubset(trades.columns):
        return {"Error": "Missing required trade columns"}

    # Sort chronologically
    trades = trades.sort_values("time").reset_index(drop=True)

    # -------- PROFIT ENGINE --------
    trades["prev_balance"] = trades["account_balance"].shift(1)
    trades["profit"] = (trades["account_balance"] - trades["prev_balance"]).round(2)
    trades["profit"] = trades["profit"].fillna(0)

    # Trade classification
    trades["result"] = "breakeven"
    trades.loc[trades["profit"] > 0, "result"] = "win"
    trades.loc[trades["profit"] < 0, "result"] = "loss"

    wins = trades[trades["result"] == "win"]
    losses = trades[trades["result"] == "loss"]
    breakeven = trades[trades["result"] == "breakeven"]

    win_count = len(wins)
    loss_count = len(losses)
    be_count = len(breakeven)
    total_trades = win_count + loss_count + be_count

    win_rate = (win_count / (win_count + loss_count) * 100) if (win_count + loss_count) else 0

    total_profit = round(wins["profit"].sum(), 2)
    total_loss = round(abs(losses["profit"].sum()), 2)
    profit_factor = round(total_profit / total_loss, 2) if total_loss else float("inf")

    avg_win = round(wins["profit"].mean(), 2) if win_count else 0
    avg_loss = round(losses["profit"].mean(), 2) if loss_count else 0

    # -------- DRAWDOWN --------
    trades["peak_equity"] = trades["account_equity"].cummax()
    trades["drawdown"] = trades["account_equity"] - trades["peak_equity"]
    max_drawdown = round(trades["drawdown"].min(), 2)

    # -------- ORDER ACTION STATS (NEW) --------
    action_stats = {}
    if "order_action" in trades.columns:
        action_counts = trades["order_action"].value_counts().to_dict()
        action_stats = {
            "Executed Trades": action_counts.get("BUY", 0) + action_counts.get("SELL", 0),
            "Skipped Trades": action_counts.get("SKIPPED_MAX_POSITIONS", 0),
            "No Trade": action_counts.get("NO_TRADE", 0)
        }

    return {
        "Total Trades": total_trades,
        "Wins": win_count,
        "Losses": loss_count,
        "Breakeven": be_count,
        "Win Rate (%)": round(win_rate, 2),
        "Balance": round(trades["account_balance"].iloc[-1], 2),
        "Equity": round(trades["account_equity"].iloc[-1], 2),
        "Net Profit": round(trades["profit"].sum(), 2),
        "Avg Win": avg_win,
        "Avg Loss": avg_loss,
        "Profit Factor": profit_factor,
        "Max Drawdown": max_drawdown,
        **action_stats
    }

def is_good_trading_day():
    day = datetime.utcnow().weekday()

    elif day in [1, 2, 3]:

def momentum_entry(price, resistance, support, signal, adx5, prev_close):
    buffer = (resistance - support) * 0.05

    if signal == "BUY":
        return (
            adx5 > 25 and
            (
                # breakout
                (prev_close > resistance and price > resistance + buffer)

                # pre-breakout
                or (price >= resistance - buffer and prev_close < resistance)

                # 🔥 continuation
                or (price > prev_close and adx5 > 30)
            )
        )

    elif signal == "SELL":
        return (
            adx5 > 25 and
            (
                (prev_close < support and price < support - buffer)
                or (price <= support + buffer and prev_close > support)
                or (price < prev_close and adx5 > 30)
            )
        )

    return False

def entry_model(
    df_5m,
    df_15m,
    signal,
    support_low,
    support_high,
    resistance_low,
    resistance_high,
    adx
):
    if df_5m is None or df_15m is None:
        return False

    if len(df_5m) < 10 or len(df_15m) < 10:
        return False

    # =========================
    # GET DATA
    # =========================
    last5 = df_5m.iloc[-1]
    prev5 = df_5m.iloc[-2]

    last15 = df_15m.iloc[-1]

    price = last5["close"]
    prev_close = prev5["close"]

    # EMAs
    ema10_5 = last5["EMA_10"]
    ema20_5 = last5["EMA_20"]

    ema10_15 = last15["EMA_10"]
    ema20_15 = last15["EMA_20"]

    # =========================
    # 🔒 1. HIGHER TIMEFRAME FILTER (15m)
    # =========================
    if signal == "BUY" and not (last15["close"] > ema10_15 > ema20_15):
        return False

    if signal == "SELL" and not (last15["close"] < ema10_15 < ema20_15):
        return False

    # =========================
    # 🔒 2. LOWER TIMEFRAME CONFIRMATION (5m)
    # =========================
    if signal == "BUY" and not (price > ema10_5 > ema20_5):
        return False

    if signal == "SELL" and not (price < ema10_5 < ema20_5):
        return False

    # =========================
    # 🔒 3. ADX (STRONGER FILTER)
    # =========================
    if adx < 20:
        return False

    # =========================
    # 🔒 4. MOMENTUM + DIRECTION
    # =========================
    body = abs(last5["close"] - last5["open"])
    candle_range = last5["high"] - last5["low"]

    if candle_range == 0:
        return False

    body_ratio = body / candle_range

    bullish = last5["close"] > last5["open"]
    bearish = last5["close"] < last5["open"]

    if body_ratio < 0.6:
        return False

    if signal == "BUY" and not bullish:
        return False

    if signal == "SELL" and not bearish:
        return False

    # =========================
    # 🔒 5. POSITION FILTER (CRITICAL)
    # =========================
    range_ = resistance_high - support_low
    if range_ <= 0:
        return False

    position = (price - support_low) / range_
    position = max(0, min(1, position))

    # ❗ Avoid trading into extremes
    if signal == "BUY" and position > 0.85:
        return False

    if signal == "SELL" and position < 0.15:
        return False

    # =========================
    # 🔒 6. STRUCTURE CONFIRMATION
    # =========================
    higher_low = last5["low"] > prev5["low"]
    lower_high = last5["high"] < prev5["high"]

    # =========================
    # 🔥 7. ENTRY LOGIC
    # =========================

    if signal == "BUY":

        # ✔ Pullback (safe zone + structure)
        pullback = (
            position < 0.5 and
            higher_low and
            price > prev_close
        )

        # ✔ Breakout + retest (REAL breakout)
        breakout = (
            prev5["close"] > resistance_high and
            last5["low"] <= resistance_high and
            price > resistance_high
        )

        if not (pullback or breakout):
            return False

    elif signal == "SELL":

        pullback = (
            position > 0.5 and
            lower_high and
            price < prev_close
        )

        breakout = (
            prev5["close"] < support_low and
            last5["high"] >= support_low and
            price < support_low
        )

        if not (pullback or breakout):
            return False

    # =========================
    # 🔒 8. ANTI-FAKEOUT FILTER
    # =========================
    wick = candle_range - body

    # large wick = rejection / trap
    if wick > body:
        return False

    return True

def delete_old_logs(log_callback=None):
    today = datetime.now().date()

    deleted_files = []

    for file in os.listdir():
        # only target bot CSV logs
        if file.startswith("logs_") and file.endswith(".csv"):
            try:
                date_str = file.replace("logs_", "").replace(".csv", "")
                file_date = datetime.strptime(date_str, "%Y-%m-%d").date()

                if file_date < today:
                    os.remove(file)
                    deleted_files.append(file)

            except Exception as e:
                if log_callback:
                    log_callback(f"⚠️ Log cleanup skip {file}: {e}")

    if log_callback and deleted_files:
        log_callback(f"🧹 Deleted old logs: {deleted_files}")
